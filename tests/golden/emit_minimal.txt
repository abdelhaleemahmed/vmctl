VBoxManage modifyvm minimal-vm --memory 2048 --vram 16 --cpus 2 --firmware bios --acpi on --ioapic off --rtcuseutc on

VBoxManage modifyvm minimal-vm --nic1 nat --nictype1 82540EM --cableconnected1 on
