VBoxManage createvm --name minimal-vm --ostype Ubuntu_64 --register

VBoxManage storagectl minimal-vm --name SATA Controller --add sata --controller IntelAhci --portcount 30 --bootable on
VBoxManage createmedium disk --filename /home/vagrant/VirtualBox VMs/minimal-vm/minimal-vm_system.vdi --size 20480 --format VDI --variant Standard
VBoxManage storageattach minimal-vm --storagectl SATA Controller --port 0 --device 0 --type hdd --medium /home/vagrant/VirtualBox VMs/minimal-vm/minimal-vm_system.vdi
VBoxManage modifyvm minimal-vm --nic1 nat --nictype1 82540EM --cableconnected1 on
VBoxManage modifyvm minimal-vm --boot1 disk
