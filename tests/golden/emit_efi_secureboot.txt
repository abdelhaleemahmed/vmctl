VBoxManage createvm --name efi-secureboot --ostype Windows11_64 --register
VBoxManage modifyvm efi-secureboot --memory 8192 --vram 128 --cpus 4 --firmware bios --acpi on --ioapic on --rtcuseutc on
VBoxManage modifyvm efi-secureboot --secureboot on
VBoxManage storagectl efi-secureboot --name SATA Controller --add sata --controller IntelAhci --portcount 2 --bootable off
VBoxManage createhd --filename /home/vagrant/VirtualBox VMs/efi-secureboot/efi-secureboot_disk_SATA Controller_0_0.vdi --size 102400 --format VDI --variant Fixed
VBoxManage storageattach efi-secureboot --storagectl SATA Controller --port 0 --device 0 --type hdd --medium /home/vagrant/VirtualBox VMs/efi-secureboot/efi-secureboot_disk_SATA Controller_0_0.vdi
VBoxManage modifyvm efi-secureboot --nic1 nat --nictype1 virtio --cableconnected1 on --macaddress1 0800274D5E6F
VBoxManage modifyvm efi-secureboot --boot1 disk
